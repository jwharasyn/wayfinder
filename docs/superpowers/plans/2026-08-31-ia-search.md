# IA Search Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A personal Internet Archive search web app: one search field, three tabs (Items / Full-text / Web), inline previews, direct download links, deployed as one Railway service.

**Architecture:** Minimal Express server proxies five archive.org endpoint families (documented advancedsearch + metadata, undocumented FTS + search-inside + Wayback anchor search) behind `/api/*` routes with an in-memory TTL cache, and serves a vanilla-JS single-page frontend from `public/`. Each undocumented backend lives in its own `lib/` module so breakage stays contained to one tab.

**Tech Stack:** Node 22 (built-in `fetch`, `node:test`), Express, vanilla JS/CSS frontend. No build step, no database, no other runtime deps.

**Spec:** `docs/superpowers/specs/2026-08-31-ia-search-design.md`

## Global Constraints

- `dns.setDefaultResultOrder("ipv4first")` first thing in `server.js` (Railway IPv6 → Cloudflare/outbound HTTPS hang fix).
- Every outbound request to archive.org sends header `User-Agent: ia-search/1.0 (personal tool)`.
- **Snippet safety rule:** snippet text from IA is untrusted and arrives in three marker formats — FTS wraps matches in `{{{...}}}`, inside.php in `<IA_FTS_MATCH>...</IA_FTS_MATCH>`, Wayback anchor in literal `<b>...</b>`. Every renderer HTML-escapes the whole string FIRST, then converts the (escaped) markers to `<mark>`/`</mark>`. Tests must include a fixture snippet containing `<script>`.
- Cache TTL 10 minutes, max 200 entries, keyed by full upstream URL. Applies to metadata lookups too (item servers move; don't cache longer).
- Each proxy route: one retry on failure/timeout, then respond `502 {"error": "<message>"}`. Frontend renders errors as a per-tab message with a retry button, never a blank screen.
- Verified endpoint URLs in this plan were probed live on 2026-08-31 — use them verbatim; do not substitute from documentation or memory.
- Desktop-first layout, responsive down to phone widths. No new tabs for previews (inline expansion); plain `<a href>` for downloads.
- ESM throughout (`"type": "module"`).
- Commit after every green test cycle. Repo is local-only until the final task (`gh repo create jwharasyn/ia-search --private`); Railway linking is John's manual step — no deploy-verify task in this plan.

export function createCache({ ttlMs = 600_000, max = 200 } = {}) {
  const map = new Map(); // insertion order = age
  return {
    get(key) {
      const e = map.get(key);
      if (!e) return undefined;
      if (Date.now() > e.expires) {
        map.delete(key);
        return undefined;
      }
      return e.value;
    },
    set(key, value) {
      if (map.size >= max && !map.has(key)) {
        map.delete(map.keys().next().value);
      }
      map.set(key, { value, expires: Date.now() + ttlMs });
    },
  };
}
```

export function escapeHtml(s) {
  return String(s)
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#39;');
}

// Escape FIRST, then convert the escaped marker forms to <mark>.
export function ftsSnippet(raw) {
  return escapeHtml(raw)
    .replaceAll('{{{', '<mark>')
    .replaceAll('}}}', '</mark>');
}

export function insideSnippet(raw) {
  return escapeHtml(raw)
    .replaceAll('&lt;IA_FTS_MATCH&gt;', '<mark>')
    .replaceAll('&lt;/IA_FTS_MATCH&gt;', '</mark>');
}

export function webSnippet(raw) {
  return escapeHtml(raw)
    .replaceAll('&lt;b&gt;', '<mark>')
    .replaceAll('&lt;/b&gt;', '</mark>');
}
```

const UA = 'ia-search/1.0 (personal tool)';

async function attempt(url, timeoutMs, fetchImpl) {
  const res = await fetchImpl(url, {
    headers: { 'User-Agent': UA, Accept: 'application/json' },
    signal: AbortSignal.timeout(timeoutMs),
  });
  if (!res.ok) throw new Error(`upstream ${res.status} from ${new URL(url).host}`);
  return res.json();
}

export async function iaFetch(url, { timeoutMs = 15_000, fetchImpl = fetch } = {}) {
  try {
    return await attempt(url, timeoutMs, fetchImpl);
  } catch {
    return attempt(url, timeoutMs, fetchImpl); // one retry; second failure propagates
  }
}
```

const FIELDS = ['identifier', 'title', 'mediatype', 'year', 'downloads', 'description'];

export function buildItemsUrl({ q, mediatype, yearFrom, yearTo, page = 1, rows = 20 }) {
  const clauses = [`(${q})`];
  if (mediatype) clauses.push(`mediatype:(${mediatype})`);
  if (yearFrom || yearTo) clauses.push(`year:[${yearFrom || '*'} TO ${yearTo || '*'}]`);
  const u = new URL('https://archive.org/advancedsearch.php');
  u.searchParams.set('q', clauses.join(' AND '));
  for (const f of FIELDS) u.searchParams.append('fl[]', f);
  u.searchParams.set('rows', String(rows));
  u.searchParams.set('page', String(page));
  u.searchParams.set('output', 'json');
  return u.toString();
}

export function reshapeItems(advJson) {
  const r = advJson.response ?? { numFound: 0, docs: [] };
  return {
    total: r.numFound,
    items: r.docs.map((d) => ({
      identifier: d.identifier,
      title: d.title ?? null,
      mediatype: d.mediatype ?? null,
      year: d.year ?? null,
      downloads: d.downloads ?? 0,
      description: Array.isArray(d.description) ? d.description.join(' ') : (d.description ?? null),
    })),
  };
}
```

dns.setDefaultResultOrder('ipv4first');

import express from 'express';
import { createCache } from './lib/cache.js';
import { iaFetch } from './lib/ia.js';
import { buildItemsUrl, reshapeItems } from './lib/items.js';

const app = express();
const cache = createCache({});

// Fetch upstream via cache; reshape; 502 on double failure.
export function proxyRoute(buildUrl, reshape) {
  return async (req, res) => {
    try {
      const url = buildUrl(req);
      let raw = cache.get(url);
      if (!raw) {
        raw = await iaFetch(url);
        cache.set(url, raw);
      }
      res.json(await reshape(raw, req));
    } catch (err) {
      res.status(502).json({ error: String(err.message ?? err) });
    }
  };
}

app.get('/api/items', proxyRoute(
  (req) => {
    if (!req.query.q) throw new Error('missing q');
    return buildItemsUrl({
      q: req.query.q,
      mediatype: req.query.mediatype || undefined,
      yearFrom: req.query.yearFrom || undefined,
      yearTo: req.query.yearTo || undefined,
      page: Number(req.query.page) || 1,
    });
  },
  (raw) => reshapeItems(raw),
));

app.use(express.static('public'));

const port = process.env.PORT || 3000;
app.listen(port, () => console.log(`ia-search listening on :${port}`));

import { ftsSnippet } from './snippets.js';

const MAX_SNIPPETS_PER_ITEM = 4;

export function buildFtsUrl({ q, page = 1, hitsPerPage = 20 }) {
  const u = new URL('https://archive.org/services/search/beta/page_production/');
  u.searchParams.set('service_backend', 'fts');
  u.searchParams.set('user_query', q);
  u.searchParams.set('page_type', 'search_results');
  u.searchParams.set('hits_per_page', String(hitsPerPage));
  u.searchParams.set('page', String(page));
  u.searchParams.set('aggregations', 'false');
  return u.toString();
}

export function reshapeFts(ftsJson) {
  const hitsObj = ftsJson.response?.body?.hits ?? { total: 0, hits: [] };
  const byId = new Map();
  for (const h of hitsObj.hits ?? []) {
    const id = h.fields?.identifier;
    if (!id) continue;
    if (!byId.has(id)) byId.set(id, { identifier: id, snippets: [] });
    const entry = byId.get(id);
    for (const raw of h.highlight?.text ?? []) {
      if (entry.snippets.length < MAX_SNIPPETS_PER_ITEM) entry.snippets.push(ftsSnippet(raw));
    }
  }
  return { total: hitsObj.total ?? 0, hits: [...byId.values()] };
}

const SKIP_NAME = /(_meta\.xml|_files\.xml|_meta\.sqlite)$/;
const SKIP_FORMAT = /Metadata|Log|Index/i;

export function reshapeItem(metaJson) {
  const id = metaJson?.metadata?.identifier ?? null;

    .filter((f) => f.name && !SKIP_NAME.test(f.name) && !SKIP_FORMAT.test(f.format ?? ''))

  return {
    identifier: id,
    title: md.title ?? null,
    description: Array.isArray(md.description) ? md.description.join(' ') : (md.description ?? null),
    mediatype: md.mediatype ?? null,
    embedUrl: `https://archive.org/embed/${id}`,
    detailsUrl: `https://archive.org/details/${id}`,

  const res = await fetch(path);
  const body = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error(body.error || `HTTP ${res.status}`);
  return body;
}

function itemsQuery(page) {
  const p = new URLSearchParams({ q: state.q, page });
  if ($('#mediatype').value) p.set('mediatype', $('#mediatype').value);
  if ($('#yearFrom').value) p.set('yearFrom', $('#yearFrom').value);
  if ($('#yearTo').value) p.set('yearTo', $('#yearTo').value);
  return `/api/items?${p}`;
}

function humanSize(n) {
  if (n == null) return '';
  const u = ['B', 'KB', 'MB', 'GB'];
  let i = 0;
  while (n >= 1024 && i < u.length - 1) { n /= 1024; i++; }
  return `${n.toFixed(i ? 1 : 0)} ${u[i]}`;
}

function makeRow({ identifier, title, metaLine, descText, snippetHtmls, expandable }) {
  const row = document.createElement('div');
  row.className = 'row';
  if (identifier) {
    const img = document.createElement('img');
    img.src = `https://archive.org/services/img/${identifier}`;
    img.loading = 'lazy';
    img.onerror = () => { img.hidden = true; };
    row.append(img);
  }
  const body = document.createElement('div');
  body.className = 'body';
  const h = document.createElement('h3');
  h.textContent = title || identifier || '(untitled)';
  const meta = document.createElement('div');
  meta.className = 'meta';
  meta.textContent = metaLine;
  body.append(h, meta);
  if (descText) {
    const d = document.createElement('p');
    d.className = 'desc';
    d.textContent = descText;
    body.append(d);
  }
  for (const html of snippetHtmls || []) {
    const s = document.createElement('p');
    s.className = 'snippet';
    s.innerHTML = html; // server-escaped snippet HTML — the only innerHTML sink
    body.append(s);
  }
  row.append(body);

  const detail = document.createElement('div');
  detail.className = 'detail';
  detail.hidden = true;

  if (expandable) {
    row.onclick = async () => {
      detail.hidden = !detail.hidden;
      if (!detail.hidden && !detail.dataset.loaded) {
        detail.dataset.loaded = '1';
        detail.textContent = 'Loading…';
        try {
          const item = await api(`/api/item/${encodeURIComponent(identifier)}`);
          renderDetail(detail, item);
          if (state.active === 'fulltext') {
            const inside = await api(`/api/inside?id=${encodeURIComponent(identifier)}&q=${encodeURIComponent(state.q)}`);
            renderInside(detail, inside);
          }
        } catch (err) {
          detail.textContent = `Couldn't load details: ${err.message}`;
        }
      }
    };
  }
  const frag = document.createDocumentFragment();
  frag.append(row, detail);
  return frag;
}

function renderDetail(detail, item) {
  detail.textContent = '';
  if (item.description) {
    const p = document.createElement('p');
    p.textContent = item.description;
    detail.append(p);
  }
  const iframe = document.createElement('iframe');
  iframe.src = item.embedUrl;
  iframe.loading = 'lazy';
  iframe.allowFullscreen = true;
  detail.append(iframe);
  const ul = document.createElement('ul');
  ul.className = 'files';
  for (const f of item.files) {
    const li = document.createElement('li');
    const a = document.createElement('a');
    a.href = f.url;
    a.textContent = f.name;
    li.append(a, ` — ${f.format ?? ''} ${humanSize(f.size)}`);
    ul.append(li);
  }
  detail.append(ul);
  const link = document.createElement('a');
  link.href = item.detailsUrl;
  link.textContent = 'on archive.org →';
  detail.append(link);
}

function renderInside(detail, inside) {
  for (const m of inside.matches) {
    const p = document.createElement('p');
    p.className = 'inside-match';
    if (m.page) {
      const pg = document.createElement('span');
      pg.className = 'pg';
      pg.textContent = `p.${m.page}`;
      p.append(pg);
    }
    const span = document.createElement('span');
    span.innerHTML = m.html; // server-escaped
    p.append(span);
    detail.append(p);
  }
}

const renderers = {
  items(pane, data, append) {
    const box = $('.results', pane);
    if (!append) box.textContent = '';
    for (const it of data.items) {
      box.append(makeRow({
        identifier: it.identifier,
        title: it.title,
        metaLine: [it.year, it.mediatype, it.downloads ? `${it.downloads} downloads` : null].filter(Boolean).join(' · '),
        descText: it.description,
        expandable: true,
      }));
    }
    const t = state.tabs.items;
    $('.more', pane).hidden = t.rows.length >= data.total || data.items.length === 0;
    if (!append && data.items.length === 0) setStatus('items', 'No results.');
  },
  fulltext(pane, data, append) {
    const box = $('.results', pane);
    if (!append) box.textContent = '';
    for (const h of data.hits) {
      box.append(makeRow({
        identifier: h.identifier,
        title: h.title,
        metaLine: [h.year, h.mediatype].filter(Boolean).join(' · '),
        snippetHtmls: h.snippets,
        expandable: true,
      }));
    }
    const t = state.tabs.fulltext;
    $('.more', pane).hidden = t.rows.length >= data.total || data.hits.length === 0;
    if (!append && data.hits.length === 0) setStatus('fulltext', 'No results.');
  },
  web(pane, data) {
    const box = $('.results', pane);
    box.textContent = '';
    for (const s of data.sites) {
      const frag = makeRow({
        identifier: null,
        title: s.name,
        metaLine: [s.firstYear && s.lastYear ? `${s.firstYear}–${s.lastYear}` : null, `${s.captures} captures`].filter(Boolean).join(' · '),
        snippetHtmls: s.snippetHtml ? [s.snippetHtml] : [],
        expandable: false,
      });
      const row = frag.firstChild;
      row.onclick = () => { location.href = s.waybackUrl; };
      box.append(frag);
    }
    if (data.sites.length === 0) setStatus('web', 'No results.');
  },
};

async function loadTab(name, force = false) {
  const t = state.tabs[name];
  if (!state.q || (t.loaded && !force)) return;
  const pane = mainPane(name);
  setStatus(name, 'Searching…');
  try {
    let data;
    if (name === 'items') data = await api(itemsQuery(t.page));
    else if (name === 'fulltext') data = await api(`/api/fulltext?q=${encodeURIComponent(state.q)}&page=${t.page}`);
    else data = await api(`/api/web?q=${encodeURIComponent(state.q)}`);
    const append = t.page > 1;
    t.loaded = true;
    if (name !== 'web') {
      t.total = data.total;
      t.rows = append ? t.rows.concat(data.items ?? data.hits) : (data.items ?? data.hits);
    }
    setStatus(name, '');
    renderers[name](pane, data, append);
  } catch (err) {
    setStatus(name, `archive.org didn't respond: ${err.message}`, true);
  }
}

function resetTabs() {
  for (const name of Object.keys(state.tabs)) {
    Object.assign(state.tabs[name], { page: 1, loaded: false, rows: [], total: 0 });
    const pane = mainPane(name);
    $('.results', pane).textContent = '';
    setStatus(name, '');
    const more = $('.more', pane);
    if (more) more.hidden = true;
  }
}

$('#search-form').onsubmit = (e) => {
  e.preventDefault();
  state.q = $('#q').value.trim();
  if (!state.q) return;
  resetTabs();
  loadTab(state.active);
};

for (const el of ['#mediatype', '#yearFrom', '#yearTo'].map((s) => $(s))) {
  el.onchange = () => {
    Object.assign(state.tabs.items, { page: 1, loaded: false, rows: [], total: 0 });
    $('.results', mainPane('items')).textContent = '';
    if (state.active === 'items') loadTab('items');
  };
}

for (const btn of document.querySelectorAll('.tab')) {
  btn.onclick = () => {
    state.active = btn.dataset.tab;
    for (const b of document.querySelectorAll('.tab')) b.classList.toggle('active', b === btn);
    for (const p of document.querySelectorAll('.pane')) p.hidden = p.dataset.pane !== state.active;
    loadTab(state.active);
  };
}

for (const name of ['items', 'fulltext']) {
  $('.more', mainPane(name)).onclick = () => {
    state.tabs[name].page += 1;
    state.tabs[name].loaded = false;
    loadTab(name);
  };
}

  ['/api/items?q=aeron+chair', (d) => d.total > 0 && d.items[0].identifier],
